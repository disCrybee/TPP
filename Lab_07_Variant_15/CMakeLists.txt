cmake_minimum_required(VERSION 3.21)
project(FileIntegrityLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(FileIntegrityLib STATIC src/FileIntegrityTask.cpp)
target_include_directories(FileIntegrityLib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(FileIntegrityDemo demo/main.cpp)
target_link_libraries(FileIntegrityDemo PRIVATE FileIntegrityLib)
add_executable(AntipatternDemo antipattern/main.cpp)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest CONFIG REQUIRED)
  add_executable(FileIntegrityTests tests/FileIntegrityTests.cpp)
  target_link_libraries(FileIntegrityTests PRIVATE FileIntegrityLib GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(FileIntegrityTests)
endif()
